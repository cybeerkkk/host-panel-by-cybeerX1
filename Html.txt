        let tokensList = [];

        function loadInitialDemoTokens() {
            if (tokensList.length === 0) {
                const sampleTokens = [
                    "otayzqxnjk1mzq6813258.gf89kl.xyz1234567890abcdefabcdef_cybeerx1",
                    "mtazmzyzyw5mtg3zqyyzqy5os5hj76yt.mnb0987654321fedcbafedcba_cybeerx1"
                ];
                sampleTokens.forEach(t => {
                    tokensList.push({
                        id: 'tok_' + Math.random().toString(36).substr(2, 9),
                        token: t,
                        status: 'unchecked',
                        vcState: 'disconnected',
                        ping: '--'
                    });
                });
                renderTokensTable();
            }
        }

        function verifyMemberKey() {
            const inputVal = document.getElementById('memberKeyInput').value.trim().toLowerCase();
            const errorMsg = document.getElementById('keyErrorMsg');

            if (inputVal === MEMBER_KEY) {
                currentRole = 'member';
                sessionStorage.setItem('cybeerx1_auth', 'true');
                sessionStorage.setItem('cybeerx1_role', 'member');
                unlockDashboard();
                showToast("access granted", "logged in as member host", "success");
            } else {
                showErrorShake(errorMsg, document.getElementById('memberKeyInput'));
            }
        }

        function verifyOwnerKey() {
            const inputVal = document.getElementById('ownerKeyInput').value.trim().toLowerCase();
            const errorMsg = document.getElementById('keyErrorMsg');

            if (inputVal === OWNER_KEY) {
                currentRole = 'owner';
                sessionStorage.setItem('cybeerx1_auth', 'true');
                sessionStorage.setItem('cybeerx1_role', 'owner');
                unlockDashboard();
                showToast("owner vip access", "logged in with full owner privileges", "success");
            } else {
                showErrorShake(errorMsg, document.getElementById('ownerKeyInput'));
            }
        }

        function showErrorShake(errorMsg, inputElem) {
            errorMsg.classList.remove('hidden');
            inputElem.classList.add('border-red-500', 'animate-shake');
            setTimeout(() => {
                inputElem.classList.remove('animate-shake');
            }, 500);
        }

        function unlockDashboard() {
            const gate = document.getElementById('keyGateModal');
            if(gate) {
                gate.style.opacity = '0';
                gate.style.pointerEvents = 'none';
                setTimeout(() => {
                    gate.style.display = 'none';
                    document.getElementById('mainDashboard').classList.remove('hidden');
                    
                    const roleNameDisplay = document.getElementById('roleNameDisplay');
                    const sessionRoleBadge = document.getElementById('sessionRoleBadge');
                    const ownerExclusiveSection = document.getElementById('ownerExclusiveSection');

                    if (currentRole === 'owner') {

        function addTokens() {
            const textarea = document.getElementById('tokenInputTextarea');
            const lines = textarea.value.split('\n').map(l => l.trim().toLowerCase()).filter(l => l.length > 0);

            if (lines.length === 0) {
                showToast("warning", "please enter at least one token", "error");
                return;

            let addedCount = 0;
            lines.forEach(tokenStr => {
                tokensList.push({

                addedCount++;
            });

            textarea.value = '';
            renderTokensTable();
            showToast("tokens imported", `successfully added ${addedCount} tokens`, "success");
        }

        function renderTokensTable(filterText = '') {
            const tbody = document.getElementById('tokensTableBody');
            const emptyState = document.getElementById('emptyTokensState');
            const tableCountBadge = document.getElementById('tableCountBadge');
            const statTotalTokens = document.getElementById('statTotalTokens');
            const statInVcTokens = document.getElementById('statInVcTokens');
            const statDeadTokens = document.getElementById('statDeadTokens');
            const globalActiveCount = document.getElementById('globalActiveCount');

            tbody.innerHTML = '';

            let filtered = tokensList;
            if (filterText) {
                filtered = tokensList.filter(t => t.token.includes(filterText.toLowerCase()));
            }

            tableCountBadge.innerText = tokensList.length;
            statTotalTokens.innerText = tokensList.length;

            let inVcCount = tokensList.filter(t => t.vcState === 'connected').length;
            let deadCount = tokensList.filter(t => t.status === 'dead').length;

            statInVcTokens.innerText = inVcCount;
            statDeadTokens.innerText = deadCount;
            globalActiveCount.innerText = inVcCount;

            if (tokensList.length === 0) {
                emptyState.classList.remove('hidden');
                return;
            } else {
                emptyState.classList.add('hidden');
            }

            filtered.forEach((item, index) => {
                let maskToken = item.token.length > 20 ? item.token.substring(0, 10) + '••••••••' + item.token.substring(item.token.length - 4) : item.token;

                let statusBadge = '';
                if (item.status === 'valid') {
                    statusBadge = `<span class="px-2 py-0.5 rounded-full bg-emerald-950/80 text-emerald-400 border border-emerald-800/60 text-[9px] font-bold">valid</span>`;
                } else if (item.status === 'dead') {
                    statusBadge = `<span class="px-2 py-0.5 rounded-full bg-red-950/80 text-red-400 border border-red-800/60 text-[9px] font-bold">dead</span>`;
                } else {
                    statusBadge = `<span class="px-2 py-0.5 rounded-full bg-amber-950/80 text-amber-400 border border-amber-800/60 text-[9px] font-bold">unchecked</span>`;
                }

                let vcBadge = '';
                if (item.vcState === 'connected') {
                    vcBadge = `<span class="px-2 py-0.5 rounded-full bg-emerald-950/60 text-emerald-300 border border-emerald-700/50 text-[9px] font-bold flex items-center gap-1 w-max"><span class="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse"></span>connected</span>`;
                } else {
                    vcBadge = `<span class="px-2 py-0.5 rounded-full bg-zinc-900 text-zinc-400 border border-zinc-800 text-[9px] font-bold w-max">disconnected</span>`;
                }

                let row = document.createElement('tr');
                row.className = "hover:bg-red-950/35 transition border-b border-red-950/40";
                row.innerHTML = `
                    <td class="py-2.5 px-3 text-red-400 font-bold">${index + 1}</td>
                    <td class="py-2.5 px-3 text-white font-mono tracking-tight" title="${item.token}">${maskToken}</td>
                    <td class="py-2.5 px-3">${statusBadge}</td>
                    <td class="py-2.5 px-3">${vcBadge}</td>
                    <td class="py-2.5 px-3 text-red-300">${item.ping}</td>
                    <td class="py-2.5 px-3 text-center">
                        <div class="flex items-center justify-center gap-1">
                            <button onclick="singleConnect('${item.id}')" title="connect to vc" class="w-6 h-6 rounded bg-emerald-950/80 hover:bg-emerald-900 text-emerald-400 border border-emerald-800/50 transition flex items-center justify-center cursor-pointer">
                                <i class="fa-solid fa-play text-[9px]"></i>

                                <i class="fa-solid fa-stop text-[9px]"></i>
                            </button>
                            <button onclick="singleCheckToken('${item.id}')" title="check token" class="w-6 h-6 rounded bg-amber-950/80 hover:bg-amber-900 text-amber-400 border border-amber-800/50 transition flex items-center justify-center cursor-pointer">
                                <i class="fa-solid fa-check text-[9px]"></i>
                            </button>
                            <button onclick="deleteSingleToken('${item.id}')" title="delete" class="w-6 h-6 rounded bg-red-950/80 hover:bg-red-900 text-red-400 border border-red-800/50 transition flex items-center justify-center cursor-pointer">

                tbody.appendChild(row);
            });

        function filterTokensTable() {
            const query = document.getElementById('tokenSearchInput').value;
            renderTokensTable(query);
        }

        function singleConnect(id) {
            const guildId = document.getElementById('guildIdInput').value.trim();
            const channelId = document.getElementById('channelIdInput').value.trim();

            if (!guildId || !channelId) {
                showToast("error", "please enter guild id and channel id first", "error");
                return;
            }

            const tokenObj = tokensList.find(t => t.id === id);
            if (tokenObj) {
                tokenObj.vcState = 'connected';
                tokenObj.status = 'valid';
                tokenObj.ping = Math.floor(Math.random() * 25 + 10) + 'ms';
                renderTokensTable();
                showToast("connected", "token successfully joined voice channel", "success");
            }
        }

        function singleDisconnect(id) {
            const tokenObj = tokensList.find(t => t.id === id);
            if (tokenObj) {
                tokenObj.vcState = 'disconnected';
                tokenObj.ping = '--';
                renderTokensTable();
                showToast("disconnected", "token left voice channel", "info");
            }
        }

        function singleCheckToken(id) {
            const tokenObj = tokensList.find(t => t.id === id);
            if (tokenObj) {
                let isValid = Math.random() > 0.1;
                tokenObj.status = isValid ? 'valid' : 'dead';
                renderTokensTable();
                showToast("check result", isValid ? "token is 100% working" : "token is dead or invalid", isValid ? "success" : "error");
            }
        }

        function massConnect() {
            const guildId = document.getElementById('guildIdInput').value.trim();
            const channelId = document.getElementById('channelIdInput').value.trim();

            if (!guildId || !channelId) {
                showToast("error", "enter guild id and channel id to mass connect", "error");
                return;
            }

            if (tokensList.length === 0) {
                showToast("warning", "no tokens available", "error");
                return;
            }

            tokensList.forEach(t => {
                t.vcState = 'connected';
                t.status = 'valid';
                t.ping = Math.floor(Math.random() * 30 + 10) + 'ms';
            });

            renderTokensTable();
            showToast("mass host", "all tokens connected to voice successfully!", "success");
        }

        function massDisconnect() {
            if (tokensList.length === 0) return;
            tokensList.forEach(t => {
                t.vcState = 'disconnected';
                t.ping = '--';
            });
            renderTokensTable();
            showToast("mass stop", "all tokens disconnected from voice", "info");
        }

        function massCheckTokens() {
            if (tokensList.length === 0) return;
            tokensList.forEach(t => {
                t.status = Math.random() > 0.15 ? 'valid' : 'dead';
            });
            renderTokensTable();
            showToast("batch check", "all tokens verified successfully", "success");
        }

        function startAntiLeaveGuard() {
            setInterval(() => {
                const antiLeaveEnabled = document.getElementById('antiLeaveCheck')?.checked;
                const guildId = document.getElementById('guildIdInput')?.value.trim();
                const channelId = document.getElementById('channelIdInput')?.value.trim();

                if (antiLeaveEnabled && guildId && channelId && tokensList.length > 0) {
                    let reconnectedAny = false;
                    tokensList.forEach(t => {
                        if (t.vcState === 'connected' && Math.random() < 0.05) {
                            reconnectedAny = true;
                            t.ping = Math.floor(Math.random() * 20 + 12) + 'ms';
                        }
                    });
                    if (reconnectedAny) {
                        renderTokensTable();
                        showToast("anti-leave guard", "token dropped from vc! auto-reconnected instantly.", "success");
                    }
                }
            }, 6000);
        }

        function showToast(title, message, type = 'success') {
            const toast = document.getElementById('toastNotification');
            const toastTitle = document.getElementById('toastTitle');
            const toastMessage = document.getElementById('toastMessage');
            const toastIconContainer = document.getElementById('toastIconContainer');
            const toastIcon = document.getElementById('toastIcon');

            toastTitle.innerText = title;
            toastMessage.innerText = message;

            if (type === 'success') {
                toastIconContainer.className = "w-8 h-8 rounded-lg bg-emerald-600/80 flex items-center justify-center text-white text-sm flex-shrink-0 shadow-md";
                toastIcon.className = "fa-solid fa-check";
            } else if (type === 'error') {
                toastIconContainer.className = "w-8 h-8 rounded-lg bg-red-600/80 flex items-center justify-center text-white text-sm flex-shrink-0 shadow-md";
                toastIcon.className = "fa-solid fa-triangle-exclamation";
            } else {
                toastIconContainer.className = "w-8 h-8 rounded-lg bg-amber-600/80 flex items-center justify-center text-white text-sm flex-shrink-0 shadow-md";
                toastIcon.className = "fa-solid fa-circle-info";
            }

            toast.classList.remove('translate-y-32', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-32', 'opacity-0');
            }, 3000);
        }
    </script>
